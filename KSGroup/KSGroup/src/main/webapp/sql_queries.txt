---------------------HOTELS TABLE------------------------
CREATE TABLE HOTELS (
    HOTEL_ID   VARCHAR2(10) NOT NULL,
    HOTEL_NAME VARCHAR2(30),
    CITY       VARCHAR2(20),
    PHOTO      VARCHAR2(255),

    CONSTRAINT PK_HOTELS PRIMARY KEY (HOTEL_ID)
);

---------------------ROOMS TABLE------------------------
CREATE TABLE ROOMS (
    HOTEL_ID VARCHAR2(10) NOT NULL,
    ROOM_ID  VARCHAR2(10) NOT NULL,
    CATEGORY VARCHAR2(15),
    PRICE    NUMBER(7,2),
    PHOTO    VARCHAR2(255),
    CAPACITY NUMBER(1,0),
    STATUS   VARCHAR2(10) NOT NULL,

    CONSTRAINT PK_ROOMS PRIMARY KEY (HOTEL_ID, ROOM_ID),

    CONSTRAINT FK_ROOMS_HOTEL
        FOREIGN KEY (HOTEL_ID)
        REFERENCES HOTELS(HOTEL_ID)
);

---------------------CUSTOMERS TABLE------------------------
CREATE TABLE CUSTOMERS (
    CUSTOMER_ID VARCHAR2(10) NOT NULL,
    FIRST_NAME  VARCHAR2(30) NOT NULL,
    MIDDLE_NAME VARCHAR2(30),
    LAST_NAME   VARCHAR2(30) NOT NULL,
    GENDER      VARCHAR2(30) NOT NULL,
    EMAIL       VARCHAR2(30) NOT NULL,
    PASSWORD    VARCHAR2(100) NOT NULL,
    DOB         DATE NOT NULL,
    PHONE       VARCHAR2(15) NOT NULL,
    PHOTO       VARCHAR2(100),

    CONSTRAINT PK_CUSTOMERS PRIMARY KEY (CUSTOMER_ID)
);

---------------------ADMIN TABLE------------------------
CREATE TABLE KSGROUP_ADMIN (
    USERNAME VARCHAR2(20),
    PASSWORD VARCHAR2(20)
);

---------------------BOOKINGS TABLE------------------------
CREATE TABLE BOOKINGS (
    BOOKING_ID   VARCHAR2(10) NOT NULL,
    CUSTOMER_ID  VARCHAR2(10) NOT NULL,
    HOTEL_ID     VARCHAR2(10) NOT NULL,
    ROOM_ID      VARCHAR2(10) NOT NULL,
    FROM_DATE    DATE NOT NULL,
    TO_DATE      DATE NOT NULL,
    NO_OF_PEOPLE NUMBER(1,0) NOT NULL,
    TOTAL_AMOUNT NUMBER(10,2) NOT NULL,
    BOOKING_DATE DATE DEFAULT SYSDATE,
    STATUS       VARCHAR2(15) DEFAULT 'BOOKED',

    CONSTRAINT PK_BOOKINGS PRIMARY KEY (BOOKING_ID),

    CONSTRAINT FK_BOOKINGS_CUSTOMER
        FOREIGN KEY (CUSTOMER_ID)
        REFERENCES CUSTOMERS(CUSTOMER_ID),

    CONSTRAINT FK_BOOKINGS_HOTEL
        FOREIGN KEY (HOTEL_ID)
        REFERENCES HOTELS(HOTEL_ID),

    CONSTRAINT FK_BOOKINGS_ROOM
        FOREIGN KEY (HOTEL_ID, ROOM_ID)
        REFERENCES ROOMS(HOTEL_ID, ROOM_ID)
);

---------------------PAYMENTS TABLE------------------------
CREATE TABLE PAYMENTS (
    BOOKING_ID   VARCHAR2(10) NOT NULL,
    CUSTOMER_ID  VARCHAR2(10) NOT NULL,
    FIRST_NAME   VARCHAR2(30) NOT NULL,
    MIDDLE_NAME  VARCHAR2(30),
    LAST_NAME    VARCHAR2(30) NOT NULL,
    PAID_ON      DATE NOT NULL,
    HOTEL_NAME   VARCHAR2(30) NOT NULL,
    PAID_AMOUNT  NUMBER(10,2) NOT NULL,
    ROOM_TYPE    VARCHAR2(15) NOT NULL,
    NO_OF_PEOPLE NUMBER(1,0) NOT NULL,
    STATUS       VARCHAR2(15) DEFAULT 'PAID',

    CONSTRAINT PK_PAYMENTS PRIMARY KEY (BOOKING_ID),

    CONSTRAINT FK_PAYMENTS_CUSTOMER
        FOREIGN KEY (CUSTOMER_ID)
        REFERENCES CUSTOMERS(CUSTOMER_ID)
);

---------------------UDB TABLE------------------------
CREATE TABLE USER_DELETED_BOOKINGS (
    CUSTOMER_ID VARCHAR2(10) NOT NULL,
    BOOKING_ID  VARCHAR2(10) NOT NULL,

    CONSTRAINT PK_USER_DELETED_BOOKINGS
        PRIMARY KEY (CUSTOMER_ID, BOOKING_ID),
);

---------------------CONTACT_MESSAGES TABLE------------------------
CREATE TABLE CONTACT_MESSAGES (
    MESSAGE_ID   VARCHAR2(10) NOT NULL,
    CUSTOMER_ID  VARCHAR2(10),
    FIRST_NAME   VARCHAR2(30) NOT NULL,
    MIDDLE_NAME  VARCHAR2(30),
    LAST_NAME    VARCHAR2(30) NOT NULL,
    MESSAGE_TEXT VARCHAR2(1000) NOT NULL,
    PARENT_ID    VARCHAR2(10),
    IS_ADMIN     VARCHAR2(1) DEFAULT 'N' NOT NULL,
    CREATED_AT   DATE DEFAULT SYSDATE NOT NULL,
    UPDATED_AT   DATE,
    PHONE        VARCHAR2(20),
    EMAIL        VARCHAR2(100),
    FULL_NAME    VARCHAR2(30),

    CONSTRAINT PK_CONTACT_MESSAGES
        PRIMARY KEY (MESSAGE_ID),

    CONSTRAINT FK_CONTACT_CUSTOMER
        FOREIGN KEY (CUSTOMER_ID)
        REFERENCES CUSTOMERS(CUSTOMER_ID)
);

---------------------MESSAGE SEQUENCE------------------------

    INCREMENT BY 1

    MAXVALUE 1000000000000000000000000000
    NOCYCLE
